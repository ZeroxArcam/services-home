     public Boolean isTransient(Object entity, SharedSessionContractImplementor session) throws HibernateException {
         Object id = this.getIdentifier(entity, session);
         if (id == null) {
             return true;
         } else {
             if (this.isVersioned()) {
                 Object version = this.getVersion(entity);
                 Boolean isUnsaved = this.versionMapping.getUnsavedStrategy().isUnsaved(version);
                 if (isUnsaved != null) {
                     if (isUnsaved) {
                         PersistenceContext persistenceContext;
                         if (version == null && (persistenceContext = session.getPersistenceContext()).hasLoadContext() && !persistenceContext.getLoadContexts().isLoadingFinished()) {
                             EntityHolder holder = persistenceContext.getEntityHolder(new EntityKey(id, this));
                             if (holder != null && holder.isEventuallyInitialized() && holder.getEntity() == entity) {
                                 return false;
                             }
                         }
 
                         Generator identifierGenerator = this.getGenerator();
                         if (identifierGenerator != null && !(identifierGenerator instanceof ForeignGenerator)) {
                             Boolean unsaved = this.identifierMapping.getUnsavedStrategy().isUnsaved(id);
                             if (unsaved != null && !unsaved) {
                                 throw new PropertyValueException("Detached entity with generated id '" + id + "' has an uninitialized version value '" + version + "'", this.getEntityName(), this.getVersionColumnName());
                             }
                         }
                     }
 
                     return isUnsaved;
                 }
             }
 
             Boolean result = this.identifierMapping.getUnsavedStrategy().isUnsaved(id);
             if (result != null) {
                 return result;
             } else {
                 if (session.getCacheMode().isGetEnabled() && this.canReadFromCache()) {
                     EntityDataAccess cache = this.getCacheAccessStrategy();
                     Object ck = cache.generateCacheKey(id, this, session.getFactory(), session.getTenantIdentifier());
                     Object ce = CacheHelper.fromSharedCache(session, ck, this, this.getCacheAccessStrategy());
                     if (ce != null) {
                         return false;
                     }
                 }
 
                 return null;
             }
         }
     }
