     private String[] doGetImplementors(Class<?> clazz) throws MappingException {
         ArrayList<String> results = new ArrayList();

             if (checkPersister instanceof Queryable) {

                 boolean isMappedClass = clazz.getName().equals(checkQueryableEntityName);
                 if (checkPersister.isExplicitPolymorphism()) {
                     if (isMappedClass) {
                         return new String[]{clazz.getName()};
                     }
                 } else if (isMappedClass) {
                     results.add(checkQueryableEntityName);
                 } else {
                     Class<?> mappedClass = checkPersister.getMappedClass();

                             String superTypeName = checkPersister.getSuperMappingType().getEntityName();
                             Class<?> mappedSuperclass = this.getEntityDescriptor(superTypeName).getMappedClass();
                             assignableSuperclass = clazz.isAssignableFrom(mappedSuperclass);
                         } else {
                             assignableSuperclass = false;
                         }
 
                         if (!assignableSuperclass) {
                             results.add(checkQueryableEntityName);
                         }
                     }
                 }
             }
         }

             return this.resolveMappingExpressible(((AnonymousTupleSqmPathSource)sqmExpressible).getSqmPathType(), tableGroupLocator);
         } else if (sqmExpressible instanceof EmbeddableTypeImpl) {
             return (MappingModelExpressible)sqmExpressible;
         } else if (sqmExpressible instanceof EntityDomainType) {
             return this.getEntityDescriptor(((EntityDomainType)sqmExpressible).getHibernateEntityName());
         } else if (!(sqmExpressible instanceof TupleType)) {
             return null;
         } else {
             MappingModelExpressible<?> mappingModelExpressible = (MappingModelExpressible)this.tupleTypeCache.get(sqmExpressible);
             if (mappingModelExpressible != null) {

             } else {

                 MappingModelExpressible<?>[] components = new MappingModelExpressible[tupleType.componentCount()];

                 MappingModelExpressible<?> existingMappingModelExpressible = (MappingModelExpressible)this.tupleTypeCache.putIfAbsent(tupleType, createdMappingModelExpressible);
                 return existingMappingModelExpressible == null ? createdMappingModelExpressible : existingMappingModelExpressible;

     public <T> BindableType<T> resolveQueryParameterType(Class<T> javaClass) {
         BasicType<T> basicType = this.getTypeConfiguration().getBasicTypeForJavaType(javaClass);
         if (basicType == null && !javaClass.isEnum()) {
             ManagedDomainType<T> managedType = this.jpaMetamodel.findManagedType(javaClass);
             if (managedType != null) {
                 return managedType;
             } else {
                 JavaTypeRegistry javaTypeRegistry = this.getTypeConfiguration().getJavaTypeRegistry();
                 JavaType<T> javaType = javaTypeRegistry.findDescriptor(javaClass);
                 if (javaType != null) {
                     JdbcType recommendedJdbcType = javaType.getRecommendedJdbcType(this.getTypeConfiguration().getCurrentBaseSqlTypeIndicators());
                     if (recommendedJdbcType != null) {
                         return this.getTypeConfiguration().getBasicTypeRegistry().resolve(javaType, recommendedJdbcType);
                     }
                 }
 
                 if (javaClass.isArray() && javaTypeRegistry.findDescriptor(javaClass.getComponentType()) != null) {
                     JavaType<T> resolvedJavaType = javaTypeRegistry.resolveDescriptor(javaClass);
                     JdbcType recommendedJdbcType = resolvedJavaType.getRecommendedJdbcType(this.getTypeConfiguration().getCurrentBaseSqlTypeIndicators());
                     if (recommendedJdbcType != null) {
                         return this.getTypeConfiguration().getBasicTypeRegistry().resolve(resolvedJavaType, recommendedJdbcType);
                     }
                 }
 
                 return null;
             }
         } else {
             return basicType;
         }
